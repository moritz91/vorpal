cmake_minimum_required(VERSION 3.25)
project(vorpal_native_dependency_audit LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This audit requires native x64 Windows/MSVC")
endif()
set(DETOURS_ROOT "" CACHE PATH "Locally built Microsoft Detours v4.0.1 checkout")
find_path(DETOURS_INCLUDE detours.h PATHS "${DETOURS_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(DETOURS_LIBRARY detours PATHS "${DETOURS_ROOT}/lib.X64" NO_DEFAULT_PATH REQUIRED)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(vorpal_native_probe64 SHARED probe.cc probe.def)
add_executable(vorpal_native_audit launcher.cc)
foreach(target vorpal_native_probe64 vorpal_native_audit)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  target_include_directories(${target} PRIVATE "${DETOURS_INCLUDE}")
  target_link_libraries(${target} PRIVATE "${DETOURS_LIBRARY}")
endforeach()
