cmake_minimum_required(VERSION 3.25)
project(vorpal_cpp_macro_observer LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 19)
  message(FATAL_ERROR "The physical observer requires LLVM 19")
endif()
set(VORPAL_CLANG_DRIVER "" CACHE FILEPATH "Matching local LLVM 19 clang-cl executable")
set(VORPAL_DIA_GUIDS "" CACHE FILEPATH "Optional replacement for an unavailable imported DIA GUID library")
if(NOT EXISTS "${VORPAL_CLANG_DRIVER}")
  message(FATAL_ERROR "Set VORPAL_CLANG_DRIVER to the matching local clang-cl executable")
endif()
if(VORPAL_DIA_GUIDS AND TARGET LLVMDebugInfoPDB)
  if(NOT EXISTS "${VORPAL_DIA_GUIDS}")
    message(FATAL_ERROR "VORPAL_DIA_GUIDS does not exist")
  endif()
  get_target_property(PDB_LIBRARIES LLVMDebugInfoPDB INTERFACE_LINK_LIBRARIES)
  list(FILTER PDB_LIBRARIES EXCLUDE REGEX "[dD][iI][aA][gG][uU][iI][dD][sS]\\.lib$")
  list(APPEND PDB_LIBRARIES "${VORPAL_DIA_GUIDS}")
  set_target_properties(LLVMDebugInfoPDB PROPERTIES INTERFACE_LINK_LIBRARIES "${PDB_LIBRARIES}")
endif()
if(LLVM_USE_CRT_RELEASE STREQUAL "MT")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
add_executable(vorpal_cpp_macro_observer observer.cc)
target_compile_features(vorpal_cpp_macro_observer PRIVATE cxx_std_17)
target_include_directories(vorpal_cpp_macro_observer PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_options(vorpal_cpp_macro_observer PRIVATE ${LLVM_DEFINITIONS_LIST})
if(NOT LLVM_ENABLE_RTTI)
  target_compile_options(vorpal_cpp_macro_observer PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/GR-> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-fno-rtti>)
endif()
file(TO_CMAKE_PATH "${VORPAL_CLANG_DRIVER}" VORPAL_CLANG_DRIVER_LITERAL)
target_compile_definitions(vorpal_cpp_macro_observer PRIVATE VORPAL_CLANG_DRIVER="${VORPAL_CLANG_DRIVER_LITERAL}")
if(TARGET clang-cpp)
  target_link_libraries(vorpal_cpp_macro_observer PRIVATE clang-cpp)
else()
  target_link_libraries(vorpal_cpp_macro_observer PRIVATE clangTooling clangFrontend clangLex clangBasic)
endif()
